cmake_minimum_required(VERSION 3.20)
project(Limestone VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
include(cmake/Dependencies.cmake)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

add_library(limestone_foundation INTERFACE)
target_include_directories(limestone_foundation INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})

add_library(limestone_parsers STATIC parsers/syntax.cpp ${limestone_ast_sources} ${limestone_ast_headers})
add_dependencies(limestone_parsers limestone-generate-parsers)
target_link_libraries(limestone_parsers PUBLIC limestone_foundation PRIVATE limestone_parser_tables limestone_dparse)
target_include_directories(limestone_parsers PUBLIC ${PROJECT_BINARY_DIR}/generated)

add_library(metacode STATIC metacode/metacode.cpp)
target_link_libraries(metacode PUBLIC limestone_foundation)
target_link_libraries(metacode PRIVATE limestone_parser_tables limestone_dparse)
target_include_directories(metacode PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/metacode)

add_library(schedrow STATIC schedrow/schedrow.cpp)
target_link_libraries(schedrow PUBLIC limestone_foundation)
target_include_directories(schedrow PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/schedrow)

add_library(regtl STATIC regtl/regtl.cpp)
target_link_libraries(regtl PUBLIC limestone_foundation)
target_include_directories(regtl PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/regtl)

add_library(limeburg STATIC limeburg/limeburg.cpp)
target_link_libraries(limeburg PUBLIC limestone_foundation schedrow)
target_include_directories(limeburg PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/limeburg)

add_library(unisel STATIC unisel/unisel.cpp)
target_link_libraries(unisel PUBLIC limestone_foundation)
target_link_libraries(unisel PUBLIC schedrow PRIVATE satie::satie)
target_include_directories(unisel PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/unisel)

add_library(bin2bin STATIC bin2bin/bin2bin.cpp)
target_link_libraries(bin2bin PUBLIC limestone_foundation metacode PRIVATE metatk::sexprtk)
if(LIMESTONE_ENABLE_PERSISTENT_CACHE)
  target_link_libraries(bin2bin PRIVATE limestone_lmdb)
endif()
target_include_directories(bin2bin PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/bin2bin)

add_library(tunah STATIC tunah/tunah.cpp tunah/specification.cpp)
target_link_libraries(tunah PUBLIC limestone_foundation)
target_link_libraries(tunah PRIVATE equinox::equinox-ng metatk::sexprtk metatk::ekippx metatk::dsltk)
target_include_directories(tunah PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/tunah)

add_library(traceml STATIC traceml/traceml.cpp traceml/pipeline_adapter.cpp)
target_link_libraries(traceml PUBLIC limestone_foundation)
target_link_libraries(traceml PRIVATE limestone_parser_tables limestone_dparse metatk::sexprtk)
target_include_directories(traceml PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/traceml)

add_library(exolayer STATIC exolayer/exolayer.cpp)
target_link_libraries(exolayer PRIVATE ExolangTk::ExtensionTk ${CMAKE_DL_LIBS})
target_include_directories(exolayer PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/exolayer)

add_library(limestone_core STATIC limestone/limestone.cpp limestone/c_api.cpp)
target_link_libraries(limestone_core PUBLIC limestone_foundation unisel regtl traceml tunah)
target_include_directories(limestone_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/limestone)

add_executable(limestone-cli limestone/main.cpp)
target_link_libraries(limestone-cli PRIVATE limestone_core metacode bin2bin)
if(BUILD_TESTING)
add_executable(limestone-parsers-test tests/parsers.cpp)
target_include_directories(limestone-parsers-test PRIVATE ${PROJECT_SOURCE_DIR}/tests)
target_link_libraries(limestone-parsers-test PRIVATE limestone_parsers limestone_core metacode)
add_test(NAME parsers COMMAND limestone-parsers-test ${PROJECT_SOURCE_DIR})
add_test(NAME syngen COMMAND ${PERL_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/syngen.pl ${PROJECT_SOURCE_DIR})
add_executable(limestone-smoke tests/smoke.cpp)
target_link_libraries(limestone-smoke PRIVATE metacode schedrow regtl limeburg unisel bin2bin tunah traceml exolayer limestone_core)
add_test(NAME limestone-smoke COMMAND limestone-smoke)
target_compile_options(limestone-smoke PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-UNDEBUG>)
foreach(suite IN ITEMS metadata selection scheduling allocation optimization traceml binary pipeline interop)
  add_executable(limestone-${suite}-test tests/${suite}.cpp)
  target_include_directories(limestone-${suite}-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/tests)
  target_link_libraries(limestone-${suite}-test PRIVATE metacode schedrow regtl limeburg unisel bin2bin tunah traceml exolayer limestone_core)
endforeach()
add_test(NAME metadata COMMAND limestone-metadata-test ${CMAKE_CURRENT_SOURCE_DIR}/metacode/infobank/isa)
foreach(suite IN ITEMS selection scheduling allocation optimization traceml pipeline)
  add_test(NAME ${suite} COMMAND limestone-${suite}-test)
endforeach()
add_test(NAME binary COMMAND limestone-binary-test ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures ${CMAKE_CURRENT_BINARY_DIR}/test-cache)
if(LIMESTONE_ENABLE_PERSISTENT_CACHE)
  target_compile_definitions(limestone-binary-test PRIVATE LIMESTONE_TEST_LMDB)
endif()
add_executable(limestone-c-api-test tests/c_api.c tests/c_header_other.c)
target_link_libraries(limestone-c-api-test PRIVATE limestone_core exolayer)
target_compile_options(limestone-c-api-test PRIVATE $<$<C_COMPILER_ID:GNU,Clang>:-UNDEBUG>)
set_target_properties(limestone-c-api-test PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES LINKER_LANGUAGE CXX)
add_test(NAME c-api COMMAND limestone-c-api-test)
add_library(limestone-test-extension MODULE tests/extension.c)
add_library(limestone-bad-extension MODULE tests/bad_extension.c)
foreach(target IN ITEMS limestone-test-extension limestone-bad-extension)
  target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/exolayer)
endforeach()
add_test(NAME interop COMMAND limestone-interop-test $<TARGET_FILE:limestone-test-extension> $<TARGET_FILE:limestone-bad-extension>)
add_executable(limestone-tunah-tuners-test tests/tunah_tuners.cpp)
target_include_directories(limestone-tunah-tuners-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/tests)
target_link_libraries(limestone-tunah-tuners-test PRIVATE tunah)
add_test(NAME tunah-tuners COMMAND limestone-tunah-tuners-test ${CMAKE_CURRENT_SOURCE_DIR}/tunah/tuners/instr-level-tune)
add_test(NAME tunah-cli COMMAND ${CMAKE_COMMAND}
  -DCLI=$<TARGET_FILE:limestone-cli> -DTUNERS=${CMAKE_CURRENT_SOURCE_DIR}/tunah/tuners/instr-level-tune
  -DBINARY=${CMAKE_CURRENT_BINARY_DIR} -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/tunah_cli.cmake)
find_program(LUA_EXECUTABLE NAMES lua lua5.4 lua5.3)
if(LUA_EXECUTABLE)
  add_test(NAME vmweave COMMAND ${CMAKE_COMMAND} -DLUA=${LUA_EXECUTABLE} -DCC=${CMAKE_C_COMPILER} -DSOURCE=${CMAKE_CURRENT_SOURCE_DIR} -DBINARY=${CMAKE_CURRENT_BINARY_DIR} -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/vmweave.cmake)

endif()

if(DUB_EXECUTABLE AND D_COMPILER)
  add_custom_target(machine-ir-test COMMAND ${DUB_EXECUTABLE} test WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/metacode/machine-ir")
  if(BUILD_TESTING)
    add_test(NAME machine-ir COMMAND ${DUB_EXECUTABLE} test)
    set_tests_properties(machine-ir PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/metacode/machine-ir")
  endif()
